import numpy as np
from scipy import ndimage


def _shift(a, dy, dx):
    h, w = a.shape
    py, px = abs(dy), abs(dx)
    p = np.pad(a, ((py, py), (px, px)), mode="reflect")
    return p[py + dy:py + dy + h, px + dx:px + dx + w]


def _normconv(z, mask, sigma):
    m = mask.astype(np.float64)
    a = ndimage.gaussian_filter(z * m, sigma, mode="reflect")
    b = ndimage.gaussian_filter(m, sigma, mode="reflect")
    return a / np.maximum(b, 1e-8)


def _jb(z, mask, guide, sig, radius, sps, k, fallback=None):
    h, w = z.shape
    r = int(radius)
    pz = np.pad(z, r, mode="reflect")
    pm = np.pad(mask.astype(np.float64), r, mode="reflect")
    pg = np.pad(guide, r, mode="reflect")
    ps = np.pad(sig, r, mode="reflect")
    num = np.zeros((h, w), dtype=np.float64)
    den = np.zeros((h, w), dtype=np.float64)
    inv_sp = 1.0 / (2.0 * sps * sps)
    inv_rg = 1.0 / (2.0 * k * k)

    for dy in range(-r, r + 1):
        ys = slice(r + dy, r + dy + h)
        for dx in range(-r, r + 1):
            xs = slice(r + dx, r + dx + w)
            diff = guide - pg[ys, xs]
            var = sig * sig + ps[ys, xs] * ps[ys, xs]
            ww = np.exp(-(dy * dy + dx * dx) * inv_sp
                        - (diff * diff) * inv_rg / np.maximum(var, 1e-12))
            ww *= pm[ys, xs]
            num += ww * pz[ys, xs]
            den += ww

    out = num / np.maximum(den, 1e-12)
    if fallback is not None:
        out = np.where(den > 1e-8, out, fallback)
    return out


def _guided(I, p, sigma, eps):
    mI = ndimage.gaussian_filter(I, sigma, mode="reflect")
    mp = ndimage.gaussian_filter(p, sigma, mode="reflect")
    cI = ndimage.gaussian_filter(I * I, sigma, mode="reflect")
    cIp = ndimage.gaussian_filter(I * p, sigma, mode="reflect")
    var = cI - mI * mI
    cov = cIp - mI * mp
    a = cov / (var + eps)
    b = mp - a * mI
    ma = ndimage.gaussian_filter(a, sigma, mode="reflect")
    mb = ndimage.gaussian_filter(b, sigma, mode="reflect")
    return ma * I + mb


def _nlm(rgb, h, search=2, patch=1):
    H, W, _ = rgb.shape
    r = search + patch
    p = np.pad(rgb, ((r, r), (r, r), (0, 0)), mode="reflect")
    num = np.zeros_like(rgb)
    den = np.zeros((H, W), dtype=np.float64)
    h2 = max(h * h, 1e-12)

    for dy in range(-search, search + 1):
        for dx in range(-search, search + 1):
            sh = p[r + dy:r + dy + H, r + dx:r + dx + W]
            d = rgb - sh
            dist = ndimage.uniform_filter(
                np.sum(d * d, axis=2), size=2 * patch + 1, mode="reflect"
            )
            ww = np.exp(-dist / h2)
            num += ww[..., None] * sh
            den += ww

    return num / np.maximum(den, 1e-8)[..., None]


def isp(raw, meta):
    raw = np.asarray(raw, dtype=np.float64)
    h, w = raw.shape

    black = float(meta.get("black_level", 0.0))
    white = float(meta.get("white_level", (1 << int(meta.get("bit_depth", 12))) - 1))
    exposure = max(float(meta.get("exposure", 1.0)), 1e-6)
    full_well = max(float(meta.get("full_well_e", 1.0)), 1.0)
    read_noise = max(float(meta.get("read_noise_e", 0.0)), 0.0)

    scale = max(white - black, 1.0)
    v = (raw - black) / scale
    z = np.clip(v / exposure, -0.35, 1.5)

    vp = np.clip(v, 0.0, 1.0)
    sig = np.sqrt(
        vp / full_well
        + (read_noise / full_well) ** 2
        + 1.0 / (12.0 * scale * scale)
    ) / exposure
    sig = np.maximum(sig, 1e-5)
    n = float(np.percentile(sig, 60.0))

    tile = np.asarray(meta["tile"], dtype=int)
    th, tw = tile.shape
    yy = np.arange(h)[:, None] % th
    xx = np.arange(w)[None, :] % tw
    idx = tile[yy, xx]

    rm = np.zeros((h, w), dtype=bool)
    gm = np.zeros((h, w), dtype=bool)
    bm = np.zeros((h, w), dtype=bool)

    for i, name in enumerate(meta.get("filter_names", [])):
        c = str(name).strip().upper()[:1]
        mm = idx == i
        if c == "R":
            rm |= mm
        elif c == "G":
            gm |= mm
        elif c == "B":
            bm |= mm

    if not gm.any():
        gm = ~(rm | bm)

    zs = ndimage.gaussian_filter(z, 0.65, mode="reflect")
    gh = 0.5 * (_shift(z, 0, 1) + _shift(z, 0, -1)) \
        + 0.25 * (2.0 * z - _shift(z, 0, 2) - _shift(z, 0, -2))
    gv = 0.5 * (_shift(z, 1, 0) + _shift(z, -1, 0)) \
        + 0.25 * (2.0 * z - _shift(z, 2, 0) - _shift(z, -2, 0))

    dh = np.abs(_shift(zs, 0, 1) - _shift(zs, 0, -1)) \
        + np.abs(2.0 * zs - _shift(zs, 0, 2) - _shift(zs, 0, -2))
    dv = np.abs(_shift(zs, 1, 0) - _shift(zs, -1, 0)) \
        + np.abs(2.0 * zs - _shift(zs, 2, 0) - _shift(zs, -2, 0))
    eps = 2.0 * n + 1e-4
    wh = 1.0 / (dh + eps)
    wv = 1.0 / (dv + eps)
    g_est = (wh * gh + wv * gv) / np.maximum(wh + wv, 1e-12)
    g_init = np.where(gm, z, g_est)

    rg = 2 if n < 0.020 else 3
    ssg = float(np.clip(0.75 + 18.0 * n, 0.75, 1.8))
    kg = float(np.clip(1.5 + 8.0 * n, 1.5, 2.2))
    g_guide = ndimage.gaussian_filter(g_init, 0.45, mode="reflect")
    g = _jb(z, gm, g_guide, sig, rg, ssg, kg, g_init)

    if n < 0.012:
        g = np.where(gm, z, g)

    rc = 2 if n < 0.020 else 3
    ssc = float(np.clip(0.9 + 16.0 * n, 0.9, 2.0))
    kc = float(np.clip(1.5 + 8.0 * n, 1.5, 2.2))
    sig_c = sig * 1.25

    def chroma(mask):
        d = np.where(mask, z - g, 0.0)
        d0 = _normconv(d, mask, max(0.75, 0.8 * ssc))
        d_full = _jb(d, mask, g, sig_c, rc, ssc, kc, d0)
        if n > 0.025:
            d_full = _guided(g, d_full, 1.5, (1.8 * n) ** 2)
        return g + d_full

    r = chroma(rm) if rm.any() else g.copy()
    b = chroma(bm) if bm.any() else g.copy()

    rgb = np.stack((r, g, b), axis=2)
    rgb = np.nan_to_num(rgb, nan=0.0, posinf=1.0, neginf=0.0)

    if n > 0.020:
        eps_l = (1.3 * n) ** 2
        eps_c = (1.9 * n) ** 2
        G = rgb[..., 1]
        g2 = _guided(G, G, 1.5, eps_l)
        r2 = _guided(G, rgb[..., 0], 1.5, eps_c)
        b2 = _guided(G, rgb[..., 2], 1.5, eps_c)
        rgb = np.stack((r2, g2, b2), axis=2)

    if n > 0.035:
        rgb = _nlm(rgb, max(0.008, 0.35 * n), search=2, patch=1)

    rgb = np.clip(rgb, 0.0, 1.0)
    srgb = np.where(
        rgb <= 0.0031308,
        12.92 * rgb,
        1.055 * np.power(np.maximum(rgb, 0.0), 1.0 / 2.4) - 0.055,
    )
    return np.clip(srgb, 0.0, 1.0).astype(np.float32)
