import numpy as np
from scipy import ndimage


def _interp(z, mask, sigma):
    m = mask.astype(np.float64)
    a = ndimage.gaussian_filter(z * m, sigma, mode="reflect")
    b = ndimage.gaussian_filter(m, sigma, mode="reflect")
    return a / np.maximum(b, 1e-8)


def _bilat(img, noise, sigma_s, k, guide=None, radius=2):
    h, w = img.shape
    p = np.pad(img, radius, mode="reflect")
    g = img if guide is None else guide
    pg = np.pad(g, radius, mode="reflect")
    out = img.copy()
    wsum = np.ones_like(img)
    sig = np.maximum(k * noise, 1e-4)
    den = 2.0 * sig * sig
    for dy in range(-radius, radius + 1):
        for dx in range(-radius, radius + 1):
            if dy == 0 and dx == 0:
                continue
            sh = p[radius + dy:radius + dy + h, radius + dx:radius + dx + w]
            gd = pg[radius + dy:radius + dy + h, radius + dx:radius + dx + w] - g
            ww = np.exp(-(dy * dy + dx * dx) / (2.0 * sigma_s * sigma_s) - gd * gd / den)
            out += ww * sh
            wsum += ww
    return out / wsum


def isp(raw, meta):
    raw = np.asarray(raw, dtype=np.float64)
    h, w = raw.shape

    black = float(meta.get("black_level", 0.0))
    white = float(meta.get("white_level", (1 << int(meta.get("bit_depth", 12))) - 1))
    exposure = max(float(meta.get("exposure", 1.0)), 1e-6)
    full_well = max(float(meta.get("full_well_e", 1.0)), 1.0)
    read_noise = float(meta.get("read_noise_e", 0.0))

    scale = max(white - black, 1.0)
    v = (raw - black) / scale
    z = v / exposure

    noise = np.sqrt(np.maximum(v, 0.0) / full_well + (read_noise / full_well) ** 2) / exposure
    nm = float(np.percentile(noise, 55.0))
    s0 = float(np.clip(0.55 + 14.0 * nm, 0.55, 1.25))
    sd = float(np.clip(0.70 + 10.0 * nm, 0.70, 1.50))
    ss = float(np.clip(0.65 + 12.0 * nm, 0.65, 1.70))

    tile = np.asarray(meta["tile"], dtype=int)
    th, tw = tile.shape
    yy = np.arange(h)[:, None] % th
    xx = np.arange(w)[None, :] % tw
    idx = tile[yy, xx]

    rm = np.zeros((h, w), dtype=bool)
    gm = np.zeros((h, w), dtype=bool)
    bm = np.zeros((h, w), dtype=bool)

    for i, name in enumerate(meta["filter_names"]):
        c = str(name).strip().upper()[:1]
        mm = idx == i
        if c == "R":
            rm |= mm
        elif c == "G":
            gm |= mm
        elif c == "B":
            bm |= mm

    if not gm.any():
        gm = ~(rm | bm)

    g0 = _interp(z, gm, s0)
    g = _bilat(g0, noise, ss, 2.0)

    if rm.any():
        d = np.where(rm, z - g, 0.0)
        r = g + _interp(d, rm, sd)
        r = _bilat(r, noise, ss * 0.9, 1.8, guide=g)
    else:
        r = g.copy()

    if bm.any():
        d = np.where(bm, z - g, 0.0)
        b = g + _interp(d, bm, sd)
        b = _bilat(b, noise, ss * 0.9, 1.8, guide=g)
    else:
        b = g.copy()

    rgb = np.stack((r, g, b), axis=2)
    rgb = np.nan_to_num(rgb, nan=0.0, posinf=1.0, neginf=0.0)
    rgb = np.clip(rgb, 0.0, 1.0)

    srgb = np.where(
        rgb <= 0.0031308,
        12.92 * rgb,
        1.055 * np.power(np.maximum(rgb, 0.0), 1.0 / 2.4) - 0.055,
    )
    return np.clip(srgb, 0.0, 1.0).astype(np.float32)
